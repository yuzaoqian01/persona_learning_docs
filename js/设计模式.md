class Command {

}

class Light {
  on() {
    console.log("灯开了");

    console.log("灯关了");

class LightOnCommand extends Command {
  constructor(light) {
    super();
    this.light = light;

  execute() {
    this.light.on();

  }

    this.command.execute();
  }

remote.pressButton();

### 发布订阅模式

```js
class EventBus {
  constructor() {
    this.events = {};
  }

  on(event, handler) {
    (this.events[event] ||= []).push(handler);
  }

  emit(event, payload) {
    (this.events[event] || []).forEach(fn => fn(payload));
  }

  off(event, handler) {
    this.events[event] = (this.events[event] || []).filter(
      fn => fn !== handler
    );
  }
}

```

```ts
export type EventMap = Record<string, any>;


type Handler<T> = (payload: T) => void | Promise<void>;

interface Listener<T> {
  handler: Handler<T>;
  once: boolean;
  priority: number;
}

export class EventBus<E extends EventMap> {
  private events = new Map<keyof E, Listener<any>[]>();

  /** 订阅 */
  on<K extends keyof E>(
    event: K,
    handler: Handler<E[K]>,
    options?: {
      once?: boolean;
      priority?: number;
    }
  ) {
    const listeners = this.events.get(event) || [];

    listeners.push({
      handler,
      once: options?.once ?? false,
      priority: options?.priority ?? 0,
    });

    // priority 高的先执行
    listeners.sort((a, b) => b.priority - a.priority);

    this.events.set(event, listeners);

    // 返回取消订阅函数（大厂常用）
    return () => this.off(event, handler);
  }

  /** 只执行一次 */
  once<K extends keyof E>(
    event: K,
    handler: Handler<E[K]>,
    priority?: number
  ) {
    return this.on(event, handler, { once: true, priority });
  }

  /** 取消订阅 */
  off<K extends keyof E>(event: K, handler: Handler<E[K]>) {
    const listeners = this.events.get(event);
    if (!listeners) return;

    this.events.set(
      event,
      listeners.filter(l => l.handler !== handler)
    );
  }

  /** 触发事件（同步） */
  emit<K extends keyof E>(event: K, payload: E[K]) {
    const listeners = this.events.get(event);
    if (!listeners) return;

    for (const l of [...listeners]) {
      l.handler(payload);
      if (l.once) {
        this.off(event, l.handler);
      }
    }
  }

  /** 触发事件（异步，常用于交易） */
  async emitAsync<K extends keyof E>(event: K, payload: E[K]) {
    const listeners = this.events.get(event);
    if (!listeners) return;

    for (const l of [...listeners]) {
      await l.handler(payload);
      if (l.once) {
        this.off(event, l.handler);
      }
    }
  }

  /** 清空某个事件 */
  clear<K extends keyof E>(event: K) {
    this.events.delete(event);
  }

  /** 清空全部 */
  clearAll() {
    this.events.clear();
  }
}

```
